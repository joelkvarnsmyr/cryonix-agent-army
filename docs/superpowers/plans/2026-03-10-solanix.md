# Solanix Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Solana DEX trading agent (Solanix) that scans established tokens with TA signals and dynamic DexScreener breakouts, executes paper trades via Jupiter, and publishes state to Redis.

**Architecture:** New `packages/agents/src/solanix/` agent extending `BaseAgent`, reusing `BrokerClient`, `StatePublisher`, `RiskManager`, `ExitMonitor`, and `PositionSizer` from Aktienix. `SolanaClient` implements `BrokerClient` using Jupiter REST API for quotes/swaps and Helius RPC for balances. Two signal sources: CoinGecko OHLCV for TA on static watchlist, DexScreener for volume-spike breakout detection.

**Tech Stack:** `@solana/web3.js ^1.95`, Jupiter v6 REST API (no SDK, native fetch), CoinGecko free API (no key), DexScreener free API (no key), Helius RPC (HELIUS_API_KEY), `@iarna/toml` (already installed), vitest (root config)

```typescript
            broker: { ...SOLANIX_DEFAULTS.broker, ...parsed.broker },
            watchlist: { ...SOLANIX_DEFAULTS.watchlist, ...parsed.watchlist },

            signals: { ...SOLANIX_DEFAULTS.signals, ...parsed.signals },
        };
    } catch {
        console.warn('[solanix] Failed to parse solanix.toml, using defaults');
        return SOLANIX_DEFAULTS;

}
```

```typescript
import type { BrokerClient, BrokerOrder, BrokerOrderResult } from '../../aktienix/brokers/types';
import type { AccountState, StockPosition } from '../../aktienix/types';
import type { SolanaPosition } from '../types';
import { randomUUID } from 'crypto';

// Solana mint addresses for the static watchlist
const MINT_ADDRESSES: Record<string, string> = {
    SOL: 'So11111111111111111111111111111111111111112',
    JUP: 'JUPyiwrYJFskUPiHa7hkeR8VUtAeFoSYbKedZNsDvCN',
    BONK: 'DezXAZ8z7PnrnRJjz3wXBoRgixCa6xjnB7YaB1pPB263',
    WIF: 'EKpQGSJtjMFqKZ9KQanSqYXRcF8fBopzLHYxdM65zcjm',
    PYTH: 'HZ1JovNiVvGrGNiiYvEozEVgZ58xaU3RKwX8eACQBCt3',
    RNDR: 'rndrizKT3MK1iimdxRdWabcF7Zg7AR5T4nud4EkHBof',
};

const USDC_MINT = 'EPjFWdd5AufqSSqeM2qN1xzybapC8G4wEGGkZwyTDt1v';
const JUPITER_QUOTE_URL = 'https://quote-api.jup.ag/v6/quote';

interface SolanaClientOptions {
    rpcUrl: string;
    paperTrading: boolean;
    initialUsdcBalance?: number;
    privateKey?: string; // base58, only needed for live mode
}

export class SolanaClient implements BrokerClient {
    private rpcUrl: string;
    private paperTrading: boolean;
    private usdcBalance: number;
    private solBalance: number;
    private positions: Map<string, SolanaPosition> = new Map();
    private connected = false;

    constructor(private opts: SolanaClientOptions) {
        this.rpcUrl = opts.rpcUrl;
        this.paperTrading = opts.paperTrading;
        this.usdcBalance = opts.initialUsdcBalance ?? 500;
        this.solBalance = 0.1; // Small SOL reserve for gas
    }

    async connect(): Promise<void> {
        if (!this.paperTrading && !this.opts.privateKey) {
            throw new Error('SOLANA_PRIVATE_KEY required for live trading');
        }
        this.connected = true;
    }

    async disconnect(): Promise<void> {
        this.connected = false;
    }

    async getAccountState(): Promise<AccountState> {
        const positionValue = Array.from(this.positions.values())
            .reduce((sum, p) => sum + p.currentPrice * p.qty, 0);

        return {
            buyingPower: this.usdcBalance,
            settledCash: this.usdcBalance,
            portfolioValue: this.usdcBalance + positionValue,
            isPatternDayTrader: false,
        };
    }

    async getPositions(): Promise<StockPosition[]> {
        return Array.from(this.positions.values()).map(p => ({
            symbol: p.symbol,
            qty: p.qty,
            avgEntryPrice: p.avgEntryPrice,
            currentPrice: p.currentPrice,
            unrealizedPl: p.unrealizedPl,
            entryTime: p.entryTime,
            highWatermark: p.highWatermark,
        }));
    }

    async submitOrder(order: BrokerOrder): Promise<BrokerOrderResult> {
        const mint = MINT_ADDRESSES[order.symbol];
        if (!mint) {
            return { success: false, message: `Unknown symbol: ${order.symbol}` };
        }

        // Get current price via Jupiter quote (1 USDC → token)
        const price = await this.getJupiterPrice(mint, order.symbol);
        if (price === null) {
            return { success: false, message: `Failed to get Jupiter quote for ${order.symbol}` };
        }

        const dollarValue = order.qty; // qty is in USD for paper mode
        const tokenQty = dollarValue / price;

        if (order.action === 'BUY') {
            if (this.usdcBalance < dollarValue) {
                return { success: false, message: 'Insufficient USDC balance' };
            }
            this.usdcBalance -= dollarValue;

            const existing = this.positions.get(order.symbol);
            if (existing) {
                // Average down/up
                const totalCost = existing.avgEntryPrice * existing.qty + dollarValue;
                const totalQty = existing.qty + tokenQty;
                existing.avgEntryPrice = totalCost / totalQty;
                existing.qty = totalQty;
                existing.currentPrice = price;
                existing.unrealizedPl = (price - existing.avgEntryPrice) * totalQty;
            } else {
                this.positions.set(order.symbol, {
                    symbol: order.symbol,
                    mintAddress: mint,
                    qty: tokenQty,
                    avgEntryPrice: price,
                    currentPrice: price,
                    unrealizedPl: 0,
                    entryTime: Date.now(),
                    highWatermark: price,
                });
            }
        } else if (order.action === 'SELL') {
            const pos = this.positions.get(order.symbol);
            if (!pos) {
                return { success: false, message: `No position in ${order.symbol}` };
            }
            const proceeds = pos.qty * price;
            this.usdcBalance += proceeds;
            this.positions.delete(order.symbol);
        }

        return {
            success: true,
            orderId: randomUUID(),
            status: 'filled',
            filledQty: tokenQty,
            filledAvgPrice: price,
        };
    }

    /** Fetch price via Jupiter quote: 1 USDC → token */
    private async getJupiterPrice(mintAddress: string, symbol: string): Promise<number | null> {
        try {
            const url = `${JUPITER_QUOTE_URL}?inputMint=${USDC_MINT}&outputMint=${mintAddress}&amount=1000000&slippageBps=50`;
            const res = await fetch(url);
            if (!res.ok) return null;
            const data = await res.json() as { outAmount?: string };
            if (!data.outAmount) return null;
            // 1 USDC (6 dec) → outAmount (token decimals vary; use raw ratio as proxy price)
            // Price = 1 / (outAmount / 1_000_000) → gives USD per token approximately
            const tokensPerUsdc = Number(data.outAmount) / 1_000_000;
            return tokensPerUsdc > 0 ? 1 / tokensPerUsdc : null;
        } catch {
            return null;
        }
    }

    /** Update current prices for all open positions (call each loop) */
    async refreshPositionPrices(): Promise<void> {
        for (const [symbol, pos] of this.positions.entries()) {
            const price = await this.getJupiterPrice(pos.mintAddress, symbol);
            if (price !== null) {
                pos.currentPrice = price;
                pos.unrealizedPl = (price - pos.avgEntryPrice) * pos.qty;
                if (price > pos.highWatermark) pos.highWatermark = price;
            }
        }
    }

    getMintAddress(symbol: string): string | undefined {
        return MINT_ADDRESSES[symbol];
    }

    addDynamicToken(symbol: string, mintAddress: string): void {
        MINT_ADDRESSES[symbol] = mintAddress;
    }
}
```

```typescript
import { calculateTechnicalIndicators, analyzeMarketStructure } from '../../../signal-validator/technical-indicators';
import type { SolanaSignal } from '../types';

// CoinGecko IDs for each symbol
const COINGECKO_IDS: Record<string, string> = {
    SOL: 'solana',
    JUP: 'jupiter-exchange-solana',
    BONK: 'bonk',
    WIF: 'dogwifcoin',
    PYTH: 'pyth-network',
    RNDR: 'render-token',
};

// Mint addresses (same as solana-client — duplicated intentionally for isolation)
const MINT_ADDRESSES: Record<string, string> = {
    SOL: 'So11111111111111111111111111111111111111112',
    JUP: 'JUPyiwrYJFskUPiHa7hkeR8VUtAeFoSYbKedZNsDvCN',
    BONK: 'DezXAZ8z7PnrnRJjz3wXBoRgixCa6xjnB7YaB1pPB263',
    WIF: 'EKpQGSJtjMFqKZ9KQanSqYXRcF8fBopzLHYxdM65zcjm',
    PYTH: 'HZ1JovNiVvGrGNiiYvEozEVgZ58xaU3RKwX8eACQBCt3',
    RNDR: 'rndrizKT3MK1iimdxRdWabcF7Zg7AR5T4nud4EkHBof',
};

interface Config {
    symbols: string[];
    bars_count: number;
    min_signal_score?: number;
}

export class SolanaSignalGenerator {
    private config: Required<Config>;

    constructor(config: Config) {
        this.config = { min_signal_score: 0.35, ...config };
    }

    async scan(): Promise<SolanaSignal[]> {
        const signals: SolanaSignal[] = [];

        for (const symbol of this.config.symbols) {
            try {
                const signal = await this.analyzeSymbol(symbol);
                if (signal && signal.action !== 'WAIT') {
                    signals.push(signal);
                }
            } catch {
                // Skip failed symbols — one failure should not block others
            }
        }

        return signals;
    }

    private async analyzeSymbol(symbol: string): Promise<SolanaSignal | null> {
        const geckoId = COINGECKO_IDS[symbol];
        if (!geckoId) return null;

        const ohlc = await this.fetchOhlc(geckoId);
        if (!ohlc || ohlc.length < 20) return null;

        const closes = ohlc.map(c => c[4]);
        const highs = ohlc.map(c => c[2]);
        const lows = ohlc.map(c => c[3]);
        const volumes = ohlc.map(() => 0); // CoinGecko OHLC endpoint lacks volume

        const currentPrice = closes[closes.length - 1];
        const ohlcvData = closes.map((close, i) => ({
            open: ohlc[i][1], high: highs[i], low: lows[i], close, volume: 0,
        }));

        const indicators = calculateTechnicalIndicators(ohlcvData as any);
        const structure = analyzeMarketStructure(ohlcvData as any);

        const score = this.computeScore(indicators, structure);
        const action = score > this.config.min_signal_score
            ? 'BUY'
            : score < -this.config.min_signal_score
            ? 'SELL'
            : 'WAIT';

        const reasoning: string[] = [];
        if (indicators.rsi < 35) reasoning.push(`RSI oversold (${indicators.rsi.toFixed(1)})`);
        if (indicators.rsi > 65) reasoning.push(`RSI overbought (${indicators.rsi.toFixed(1)})`);
        if (indicators.macd > indicators.macdSignal) reasoning.push('MACD bullish cross');
        if (indicators.macd < indicators.macdSignal) reasoning.push('MACD bearish cross');

        return {
            symbol,
            mintAddress: MINT_ADDRESSES[symbol] ?? '',
            action,
            confidence: Math.min(Math.abs(score), 1.0),
            reasoning: reasoning.length > 0 ? reasoning : [`Score: ${score.toFixed(2)}`],
            currentPrice,
            source: 'ta',
            timestamp: Date.now(),
        };
    }

    private computeScore(indicators: any, structure: any): number {
        let score = 0;

        // RSI component (weight 0.35)
        if (indicators.rsi < 35) score += 0.35;
        else if (indicators.rsi < 45) score += 0.15;
        else if (indicators.rsi > 65) score -= 0.35;
        else if (indicators.rsi > 55) score -= 0.15;

        // MACD component (weight 0.30)
        if (indicators.macd > indicators.macdSignal) score += 0.30;
        else score -= 0.30;

        // Bollinger component (weight 0.20)
        if (indicators.bollingerPosition === 'below_lower') score += 0.20;
        else if (indicators.bollingerPosition === 'near_lower') score += 0.10;
        else if (indicators.bollingerPosition === 'above_upper') score -= 0.20;
        else if (indicators.bollingerPosition === 'near_upper') score -= 0.10;

        // Trend component (weight 0.15)
        if (structure?.trend === 'uptrend') score += 0.15;
        else if (structure?.trend === 'downtrend') score -= 0.15;

        return score;
    }

    private async fetchOhlc(geckoId: string): Promise<number[][] | null> {
        try {
            const url = `https://api.coingecko.com/api/v3/coins/${geckoId}/ohlc?vs_currency=usd&days=7`;
            const res = await fetch(url);
            if (!res.ok) return null;
            return await res.json() as number[][];
        } catch {
            return null;
        }
    }

    /** Add a dynamic symbol (from DexScreener) to the watchlist temporarily */
    addSymbol(symbol: string, mintAddress: string, geckoId?: string): void {
        if (!this.config.symbols.includes(symbol)) {
            this.config.symbols.push(symbol);
        }
        MINT_ADDRESSES[symbol] = mintAddress;
        if (geckoId) COINGECKO_IDS[symbol] = geckoId;
    }

    removeSymbol(symbol: string): void {
        this.config.symbols = this.config.symbols.filter(s => s !== symbol);
    }
}
```

```typescript
import type { DexScreenerToken } from '../types';

interface ScannerConfig {
    minLiquidityUsd: number;
    volumeSpikeMultiplier: number;
    dynamicTtlMs: number;
}

interface DynamicEntry {
    symbol: string;
    mintAddress: string;
    priceUsd: number;
    addedAt: number;
}

export class DexScreenerScanner {
    private config: ScannerConfig;
    private dynamicTokens: Map<string, DynamicEntry> = new Map();

    constructor(config: ScannerConfig) {
        this.config = config;
    }

    async scan(): Promise<DexScreenerToken[]> {
        try {
            // Search for top Solana pairs by volume
            const res = await fetch('https://api.dexscreener.com/latest/dex/tokens/So11111111111111111111111111111111111111112');
            if (!res.ok) return [];

            const data = await res.json() as { pairs?: any[] };
            if (!data.pairs) return [];

            return data.pairs
                .filter(pair => this.passesFilters(pair))
                .map(pair => this.mapToken(pair));
        } catch {
            return [];
        }
    }

    /** Scan a specific token by mint address */
    async scanToken(mintAddress: string): Promise<DexScreenerToken | null> {
        try {
            const res = await fetch(`https://api.dexscreener.com/latest/dex/tokens/${mintAddress}`);
            if (!res.ok) return null;
            const data = await res.json() as { pairs?: any[] };
            if (!data.pairs?.[0]) return null;
            return this.mapToken(data.pairs[0]);
        } catch {
            return null;
        }
    }

    private passesFilters(pair: any): boolean {
        const liq = pair.liquidity?.usd ?? 0;
        const volH1 = pair.volume?.h1 ?? 0;
        const volH24 = pair.volume?.h24 ?? 0;
        const avgHourly = volH24 / 24;
        const spikeRatio = avgHourly > 0 ? volH1 / avgHourly : 0;

        return (
            liq >= this.config.minLiquidityUsd &&
            spikeRatio >= this.config.volumeSpikeMultiplier
        );
    }

    private mapToken(pair: any): DexScreenerToken {
        const volH24 = pair.volume?.h24 ?? 0;
        const volH1 = pair.volume?.h1 ?? 0;
        const avgHourly = volH24 / 24;

        return {
            symbol: pair.baseToken?.symbol ?? 'UNKNOWN',
            mintAddress: pair.baseToken?.address ?? '',
            priceUsd: parseFloat(pair.priceUsd ?? '0'),
            volume24h: volH24,
            volumeChange1h: avgHourly > 0 ? volH1 / avgHourly : 0,
            liquidityUsd: pair.liquidity?.usd ?? 0,
            pairCreatedAt: pair.pairCreatedAt ?? 0,
        };
    }

    addToken(symbol: string, mintAddress: string, priceUsd: number): void {
        this.dynamicTokens.set(symbol, {
            symbol, mintAddress, priceUsd, addedAt: Date.now(),
        });
    }

    evictExpired(): void {
        const now = Date.now();
        for (const [symbol, entry] of this.dynamicTokens.entries()) {
            if (now - entry.addedAt >= this.config.dynamicTtlMs) {
                this.dynamicTokens.delete(symbol);
            }
        }
    }

    getDynamicSymbols(): string[] {
        return Array.from(this.dynamicTokens.keys());
    }

    getDynamicTokens(): DynamicEntry[] {
        return Array.from(this.dynamicTokens.values());
    }
}
```

```typescript
import { BaseAgent, type AgentConfig } from '../base-agent';
import type { AgentMessage } from '@cryonix/types';
import { SolanaClient } from './brokers/solana-client';
import { SolanaSignalGenerator } from './signals/solana-signal-generator';
import { DexScreenerScanner } from './signals/dexscreener-scanner';
import { StatePublisher } from '../aktienix/state-publisher';
import { RiskManager } from '../aktienix/risk-manager';
import { PositionSizer } from '../aktienix/position-sizer';
import { loadSolanixConfig, type SolanixTomlConfig } from './config';

export class SolanaAgent extends BaseAgent {
    private config!: SolanixTomlConfig;
    private brokerClient!: SolanaClient;
    private signalGenerator!: SolanaSignalGenerator;
    private dexScanner!: DexScreenerScanner;
    private riskManager!: RiskManager;
    private statePublisher!: StatePublisher;
    private isRunning = false;
    private lastSignalScan = 0;
    private lastDexScan = 0;

    constructor(config: AgentConfig) {
        super(config);
    }

    async initialize(): Promise<void> {
        this.log('info', 'Initializing Solanix Agent (Solana DEX Trading)...');
        this.config = loadSolanixConfig();

        this.riskManager = new RiskManager({
            max_daily_loss_pct: this.config.risk.max_daily_loss_pct,
            max_weekly_loss_pct: this.config.risk.max_weekly_loss_pct,
            max_consecutive_losses: this.config.risk.max_consecutive_losses,
            max_open_positions: this.config.limits.max_open_positions,
            max_trades_per_day: this.config.limits.max_trades_per_day,
            min_confidence: this.config.risk.min_confidence,
            min_settled_cash: this.config.capital.min_order_value,
            max_day_trades_per_5d: 999, // No PDT rule on Solana
        });

        this.statePublisher = new StatePublisher(
            process.env.KV_URL ?? 'redis://localhost:6379',
            'solanix', // namespace prefix
        );

        this.brokerClient = new SolanaClient({
            rpcUrl: process.env.SOLANA_RPC_URL ?? 'https://api.mainnet-beta.solana.com',
            paperTrading: this.config.broker.paper_trading,
            initialUsdcBalance: this.config.capital.max_position_size_usd * this.config.limits.max_open_positions,
            privateKey: process.env.SOLANA_PRIVATE_KEY,
        });

        this.signalGenerator = new SolanaSignalGenerator({
            symbols: this.config.watchlist.symbols,
            bars_count: this.config.watchlist.bars_count,
            min_signal_score: this.config.signals.min_signal_score,
        });

        this.dexScanner = new DexScreenerScanner({
            minLiquidityUsd: this.config.signals.dexscreener_min_liquidity_usd,
            volumeSpikeMultiplier: this.config.signals.dexscreener_volume_spike_multiplier,
            dynamicTtlMs: this.config.signals.dexscreener_dynamic_ttl_ms,
        });

        await this.brokerClient.connect();
        const account = await this.brokerClient.getAccountState();
        this.riskManager.setPortfolioValue(account.portfolioValue);

        this.setStatus('idle');
        this.log('info', `Solanix initialized. Paper mode: ${this.config.broker.paper_trading}. Balance: $${account.portfolioValue}`);
    }

    async start(): Promise<void> {
        this.log('info', 'Starting Solanix trading loop...');
        this.setStatus('running');
        this.isRunning = true;

        while (this.isRunning) {
            try {
                await this.runLoop();
            } catch (err: any) {
                this.log('error', `Loop error: ${err.message}`);
            }
            await this.sleep(this.config.timing.poll_interval_ms);
        }
    }

    async stop(): Promise<void> {
        this.log('info', 'Stopping Solanix Agent...');
        this.isRunning = false;
        await this.brokerClient.disconnect();
        await this.statePublisher.publishStatus('stopped');
        await this.statePublisher.disconnect();
        this.setStatus('paused');
    }

    async handleMessage(message: AgentMessage): Promise<void> {
        if (message.message_type === 'TRADE_SIGNAL') {
            await this.evaluateSignal(message.data);
        }
    }

    private async runLoop(): Promise<void> {
        const now = Date.now();

        // Check circuit breaker
        const circuitBreakerActive = await this.statePublisher.get('solanix:circuit_breaker');
        if (circuitBreakerActive) {
            this.log('warn', 'Circuit breaker active — skipping loop');
            await this.statePublisher.publishStatus('paused');
            return;
        }

        // Refresh account + positions
        await this.brokerClient.refreshPositionPrices();
        const account = await this.brokerClient.getAccountState();
        const positions = await this.brokerClient.getPositions();
        this.riskManager.setPortfolioValue(account.portfolioValue);

        this.log('info', `Portfolio: $${account.portfolioValue.toFixed(2)} | Positions: ${positions.length}`);

        // Risk gate
        const riskCheck = this.riskManager.check();
        if (!riskCheck.canTrade) {
            this.log('warn', `Risk gate: ${riskCheck.reason}`);
            await this.statePublisher.publishStatus('paused');
            return;
        }

        // DexScreener scan (every dexscreener_scan_interval_ms)
        if (this.config.signals.dexscreener_enabled &&
            now - this.lastDexScan >= this.config.timing.dexscreener_scan_interval_ms) {

            this.dexScanner.evictExpired();
            const breakouts = await this.dexScanner.scan();
            for (const token of breakouts) {
                this.log('info', `DexScreener breakout: ${token.symbol} (+${token.volumeChange1h.toFixed(1)}x vol)`);
                this.brokerClient.addDynamicToken(token.symbol, token.mintAddress);
                this.signalGenerator.addSymbol(token.symbol, token.mintAddress);
                this.dexScanner.addToken(token.symbol, token.mintAddress, token.priceUsd);
            }
            this.lastDexScan = now;
        }

        // TA signal scan (every signal_scan_interval_ms)
        if (now - this.lastSignalScan >= this.config.timing.signal_scan_interval_ms) {
            const signals = await this.signalGenerator.scan();
            for (const signal of signals) {
                await this.evaluateSignal(signal);
            }
            this.lastSignalScan = now;
        }

        // Publish state
        await this.statePublisher.publishPositions(positions);
        await this.statePublisher.publishAccount(account);
        await this.statePublisher.publishRisk(this.riskManager.getState());
        await this.statePublisher.publishStatus('running');
    }

    private async evaluateSignal(payload: any): Promise<void> {
        const { symbol, action, confidence, currentPrice } = payload;
        if (!symbol || action !== 'BUY') return; // Phase 1: BUY only

        if (confidence < this.config.risk.min_confidence) {
            this.log('info', `Signal ${symbol} confidence ${confidence} below threshold. Skip.`);
            return;
        }

        const riskCheck = this.riskManager.check();
        if (!riskCheck.canTrade) return;

        const account = await this.brokerClient.getAccountState();
        const dollarAmount = PositionSizer.calc(
            currentPrice ?? 1,
            account.settledCash,
            this.config.capital,
        );

        if (dollarAmount <= 0) {
            this.log('warn', `Position sizer returned 0 for ${symbol}`);
            return;
        }

        this.log('info', `BUY ${symbol} ~$${dollarAmount.toFixed(2)} (confidence: ${confidence})`);
        const result = await this.brokerClient.submitOrder({
            symbol, action: 'BUY', qty: dollarAmount, orderType: 'MARKET',
        });

        if (result.success) {
            this.riskManager.recordTradeResult(0, false);
            await this.statePublisher.addSignal({ symbol, action, confidence, reasoning: '', timestamp: Date.now() });
            this.log('info', `Order filled: ${result.orderId}`);
        } else {
            this.log('error', `Order failed: ${result.message}`);
        }
    }
}
```

```typescript
import { NextResponse } from 'next/server';
import { redis } from '@/lib/redis';

export const dynamic = 'force-dynamic';

interface AccountState {
  usdcBalance: number;
  solBalance: number;
  portfolioValue: number;
}

interface RiskState {
  canTrade: boolean;
  dailyPnl: number;
  consecutiveLosses: number;
  tradesToday: number;
}

export async function GET() {
  let agentStatus: string | null = null;
  let account: AccountState | null = null;
  let risk: RiskState | null = null;

  try {
    const [statusRaw, accountRaw, riskRaw] = await Promise.all([
      redis.get('solanix:status'),
      redis.get('solanix:account'),
      redis.get('solanix:risk'),
    ]);

    agentStatus = (statusRaw as string) ?? null;
    if (accountRaw) { try { account = JSON.parse(accountRaw as string); } catch {} }
    if (riskRaw) { try { risk = JSON.parse(riskRaw as string); } catch {} }
  } catch (error) {
    console.error('[solanix/status] Redis error:', error);
  }

  return NextResponse.json({
    status: agentStatus,
    account,
    risk,
    timestamp: new Date().toISOString(),
  });
}
```
